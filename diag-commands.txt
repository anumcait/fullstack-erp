CMD 1 — ALL-IN-ONE health check (run this one)
------------------------------------------------------------------------
Tells us WHY no containers exist: docker, compose, repo clone, compose file,
service state, and the bootstrap log — in a single ASCII-safe call.

CMD=$(aws ssm send-command \
  --instance-id i-01d05324adefba29b \
  --document-name AWS-RunShellScript \
  --parameters 'commands=["echo DOCKER=; docker --version","echo COMPOSE=; docker compose version","echo REPO=; ls /opt/erp-app 2>&1 | head -20","echo CFGFILE=; test -f /opt/erp-app/docker-compose.demo.yaml && echo EXISTS || echo MISSING","echo SVC=; systemctl is-active erp-demo.service; systemctl show erp-demo.service -p SubState -p Result","echo BOOT=; tail -n 40 /var/log/erp-bootstrap.log"]' \
  --query "Command.CommandId" --output text); \
sleep 8; \
aws ssm get-command-invocation --command-id $CMD --instance-id i-01d05324adefba29b --query StandardOutputContent --output text

WHAT TO LOOK FOR:
  - "COMPOSE=" empty or "compose is not a command"  => docker-compose-plugin MISSING (prime suspect)
  - "REPO=" shows "No such file or directory"         => git clone FAILED (wrong repo_url / private repo)
  - "SYSTEMD=" shows "active (running)" but no docker => compose up exited immediately
